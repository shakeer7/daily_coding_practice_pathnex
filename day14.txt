---

- name: Install nginx on server
  hosts: all
  become: true

  tasks:
    - name: install nginx
      apt:
        image: nginx
        state: present
    - name: start nginx server
      server:
        name: nginx
        state: started
        enable: yes
    - name: Deploy html
      copy: 
        content: |
          <!DOCTYPE html>
          <html>
          <head>
            <title>My Ansible Server</title>
          </head>
          <body>
            <h1>Hello from Ansible!</h1>
          </body>
          </html>
        dest: /var/www/html/index.html
        owner: root
        group: root
        mode: '066'
